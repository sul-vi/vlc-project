```python
    train_snr: tuple = (0., 25.)

    mode_probs: tuple = (0.25, 0.25, 0.50)  # 分别对应 2、4、8 阶

        low, high = self.thresholds
        return 2 if snr < low else 4 if snr < high else 8

    def directory(self, name):
        return ROOT / name / self.run


def parameters(description, training):
    """读取公共参数与当前入口支持的覆盖项。"""
    defaults = Config()
    parser = argparse.ArgumentParser(description=description)
    parser.add_argument("--dataset", choices=("cifar10", "folder", "synthetic"))
    parser.add_argument("--device", choices=("auto", "cpu", "mps", "cuda"))
    parser.add_argument("--run", help="checkpoints 和 results 下的实验名")
    parser.add_argument("--data-dir")
    if not training:
        parser.add_argument("--output-dir", help="评估输出位置，默认 results/<run>")
    keys = ["batch_size", "seed", "image_size"]
    keys += ["epochs", "train_limit", "val_limit"] if training else ["test_limit"]
    for key in keys:
        parser.add_argument("--" + key.replace("_", "-"), type=int)
    args = {k: v for k, v in vars(parser.parse_args()).items() if v is not None}
    cfg = Config(**args)

        parser.error("--run 必须是单个目录名")
    if min(cfg.batch_size, cfg.epochs, cfg.channels, cfg.width) < 1:
        parser.error("batch size、轮数和网络宽度必须为正数")
    if min(cfg.train_limit, cfg.val_limit, cfg.test_limit, cfg.image_size, cfg.seed) < 0:
        parser.error("样本上限、图像尺寸和种子须为非负数")
    if cfg.dataset == "folder" and cfg.image_size == 0 and cfg.batch_size != 1:
        parser.error("文件夹原尺寸模式请使用 --batch-size 1，以支持不同尺寸")
    if not 0 <= cfg.thresholds[0] < cfg.thresholds[1]:

    torch.backends.cudnn.benchmark = False
    torch.backends.cudnn.deterministic = True
    torch.use_deterministic_algorithms(True, warn_only=True)
    device = cfg.device
    if device == "auto":
        device = "cuda" if torch.cuda.is_available() else "mps" if torch.backends.mps.is_available() else "cpu"
    if device == "cpu":
        torch.set_num_threads(min(8, torch.get_num_threads()))
    return torch.device(device)
```

## channel/mcpam_channel.py

```python
"""RGB 无滤光片、等效响应度为 1 的单 PD 链路。分支轴固定按 R、B、G 排列。"""
import math
from dataclasses import dataclass

import torch
from torch import nn


@dataclass
class Reception:
    bits: torch.Tensor          # [N,3,S]，关闭分支输出 0
    weights: torch.Tensor       # 活动光功率比例，总和为 1
    optical_mean: torch.Tensor  # [N]，LED 模型后实际发送的平均光强


class MCPAMChannel(nn.Module):
    """每个信息字发送原码与反码两个时隙，保证每帧平均光功率严格一致。"""

    def __init__(self, ratios=(6., 2., 1.), compression=0.8, balanced_pairs=True):
        super().__init__()
        if len(ratios) != 3 or not all(math.isfinite(x) for x in ratios):
            raise ValueError("需要三个有限的 R/B/G 权重")
        r, b, g = ratios
        if not (r > b + g and b > g > 0):
            raise ValueError("逐层判决要求 R > B + G 且 B > G > 0")
        if not math.isfinite(compression) or compression < 0:
            raise ValueError("LED 压缩系数必须有限且非负")
        self.register_buffer("ratios", torch.tensor(ratios, dtype=torch.float32))
        self.compression = compression
        self.slots_per_word = 2 if balanced_pairs else 1

    def weights(self, mode):
        """将本阶活动颜色的峰值光功率之和归一到 1。"""
        if mode not in (2, 4, 8):
            raise ValueError("mode 取 2、4 或 8")
        weights = self.ratios.clone()
        weights[int(math.log2(mode)):] = 0
        return weights / weights.sum()

    def led(self, x):
        """等效公共驱动的中点对称软限幅，f(0)=0、f(1)=1、f(1-x)=1-f(x)。"""
        k = self.compression
        if k == 0:
            return x
        return 0.5 + torch.tanh(k * (2 * x - 1)) * (0.5 / math.tanh(k))

    @staticmethod
    def noise_sigma(snr_db):
        """以单位响应度和平均光强 0.5 为参考，返回单时隙电流噪声标准差。"""
        return 0.5 * 10.0 ** (-snr_db / 20)

    def sic(self, received, weights, levels):
        """在 PD 电流域比较相邻子星座边界，在发射幅度域累计已判分量。"""
        prefix = torch.zeros_like(received)
        residual = received
        base = self.led(prefix)
        decoded = []
        for index in range(levels):
            weight = weights[index]
            tail = weights[index + 1:].sum()
            # 当低位取全 1 或全 0 时，两簇相邻端点经 LED 映射后取中点。
            threshold = (self.led(prefix + tail) + self.led(prefix + weight)) * 0.5 - base
            bit = (residual >= threshold).to(received.dtype)
            decoded.append(bit)
            prefix = prefix + weight * bit
            # 线性时这一步就是 residual -= weight * bit；软限幅时扣除标定的光强差。
            next_base = self.led(prefix)
            residual = residual - (next_base - base)
            base = next_base
        decoded.extend(torch.zeros_like(received) for _ in range(3 - levels))
        return torch.stack(decoded, dim=1)

    def forward(self, bits, snr_db, mode=8, noise=None):
        """输入浮点二进制 [N,3,S]，snr_db 为标量。noise 可传标准正态 [N,时隙数,S]。"""
        if bits.ndim != 3 or bits.shape[1] != 3 or not bits.is_floating_point():
            raise ValueError("输入形状须为 [批量,3,每色比特数]，元素为浮点 0/1")
        weights = self.weights(mode)
        x = (bits * weights.view(1, 3, 1)).sum(dim=1)
        pair = self.slots_per_word == 2
        clean = self.led(torch.stack((x, 1 - x), dim=1) if pair else x.unsqueeze(1))
        if noise is None:
            noise = torch.randn_like(clean)
        if noise.shape != clean.shape:
            raise ValueError("噪声必须与实际发送时隙的形状一致")
        received = clean + self.noise_sigma(snr_db) * noise
        combined = (received[:, 0] + 1 - received[:, 1]) * 0.5 if pair else received[:, 0]
        hard = self.sic(combined, weights, int(math.log2(mode)))
        if self.training:
            # 按数据流作硬判决 STE。前向保留真实 SIC 错误，反向对活动位用恒等近似。
            # 直接微分离散 SIC 会给出零梯度；停用颜色不能接收梯度。
            surrogate = bits * (weights > 0).view(1, 3, 1)
            hard = surrogate + (hard - surrogate).detach()
        return Reception(hard, weights, clean.mean(dim=(1, 2)))
```

## models/semantic_net.py

```python
"""全卷积语义编解码器。通信分支 R/B/G 与输入图像 RGB 像素轴互相独立。"""
import torch
import torch.nn.functional as F
from torch import nn


class ResidualBlock(nn.Module):
    def __init__(self, width):
        super().__init__()
        self.body = nn.Sequential(
            nn.Conv2d(width, width, 3, padding=1),
            nn.GroupNorm(1, width),
            nn.SiLU(),
            nn.Conv2d(width, width, 3, padding=1),
        )

    def forward(self, x):
        return F.silu(x + self.body(x))


class SemanticNet(nn.Module):
    """每色占 C×ceil(H/2)×ceil(W/2) 个信息位，解码后裁回原尺寸。"""

    def __init__(self, channels=8, width=32):
        super().__init__()
        self.channels = channels
        self.encoder = nn.Sequential(
            nn.Conv2d(3, width, 5, stride=2, padding=2),
            nn.SiLU(), ResidualBlock(width),
            nn.Conv2d(width, 3 * channels, 1),
        )
        self.decoder = nn.Sequential(
            nn.Conv2d(3 * channels, width, 3, padding=1),
            nn.SiLU(), ResidualBlock(width),
            nn.Conv2d(width, 4 * width, 3, padding=1),
            nn.PixelShuffle(2), nn.SiLU(),
            nn.Conv2d(width, 3, 3, padding=1), nn.Sigmoid(),
        )

    def encode(self, image):
        """输入 [N,3,H,W] 且像素在 [0,1]。返回 R/B/G 位流及特征空间尺寸。"""
        logits = self.encoder(image)
        soft = torch.sigmoid(logits)
        hard = (logits >= 0).to(logits.dtype)
        quantized = soft + (hard - soft).detach() if self.training else hard
        return quantized.reshape(image.size(0), 3, -1), logits.shape[-2:]

    def decode(self, bits, latent_size, output_size):
        h, w = output_size
        features = bits.reshape(bits.size(0), 3 * self.channels, *latent_size)
        return self.decoder(features)[:, :, :h, :w]

    def forward(self, image, link, snr_db, mode=8, noise=None):
        bits, shape = self.encode(image)
        received = link(bits, snr_db, mode, noise)
        reconstruction = self.decode(received.bits, shape, image.shape[-2:])
        return reconstruction, bits, received
```

## baselines/jpeg_mcpam.py

```python
"""预算内的完整 JPEG 码流。像素按 RGB，连续比特按 R、B、G 轮流装入信息字。"""
import io
from dataclasses import dataclass

import numpy as np
import torch
from PIL import Image


@dataclass
class JPEGFrames:
    bits: torch.Tensor
    sizes: list        # 含 JPEG 头部的实际字节数，0 表示发端预算不足


def pack(images, symbols, qualities):
    """symbols 是每张图的信息字数 S，预算 floor(3S/8) 字节，空位补零。"""
    arrays = images.detach().clamp(0, 1).mul(255).round().byte().permute(0, 2, 3, 1).cpu().numpy()
    frames = np.zeros((len(arrays), symbols * 3), dtype=np.uint8)
    sizes = []
    for index, array in enumerate(arrays):
        image = Image.fromarray(array)
        size = 0
        for quality in qualities:
            stream = io.BytesIO()
            image.save(stream, format="JPEG", quality=quality, subsampling=2, optimize=True)
            payload = stream.getvalue()
            if len(payload) * 8 <= symbols * 3:
                size = len(payload)
                frames[index, :size * 8] = np.unpackbits(np.frombuffer(payload, np.uint8))
                break
        sizes.append(size)
    bits = torch.from_numpy(frames.reshape(-1, symbols, 3)).transpose(1, 2)
    return JPEGFrames(bits.to(device=images.device, dtype=images.dtype), sizes)


def unpack(bits, image_size):
    """接收端只使用固定帧预算和约定尺寸。无法解码时输出中灰，并计入图像指标。"""
    h, w = image_size
    flat = bits.transpose(1, 2).contiguous().byte().cpu().numpy().reshape(bits.shape[0], -1)
    payloads = np.packbits(flat[:, :flat.shape[1] // 8 * 8], axis=1)
    images = []
    failures = 0
    for payload in payloads:
        try:
            with Image.open(io.BytesIO(payload.tobytes()), formats=("JPEG",)) as image:
                if image.size != (w, h):
                    raise ValueError("帧尺寸错误")
                decoded = np.array(image.convert("RGB"), dtype=np.float32) / 255
        except (OSError, SyntaxError, ValueError, Image.DecompressionBombError):
            decoded = np.full((h, w, 3), 0.5, dtype=np.float32)
            failures += 1
        images.append(decoded)
    return torch.from_numpy(np.stack(images)).permute(0, 3, 1, 2).to(bits.device), failures
```

## data.py

```python
"""图像源与固定数据划分。CIFAR-10 直接读取官方二进制归档，无需反序列化 pickle。"""
import hashlib
import os
import shutil
import tarfile
import tempfile
import urllib.request
from pathlib import Path

import numpy as np
import torch
from PIL import Image, ImageOps
from torch.utils.data import Dataset, Subset

CIFAR_URL = "https://cave.cs.toronto.edu/kriz/cifar-10-binary.tar.gz"
CIFAR_MD5 = "c32a1d4ab5d03f1284b67883e8d87530"


def checksum(path):
    digest = hashlib.md5()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1 << 20), b""):
            digest.update(block)
    return digest.hexdigest()


def cifar_archive(root):
    """仅在官方 MD5 匹配后原子发布归档。中断下载不会变成有效数据集。"""
    root.mkdir(parents=True, exist_ok=True)
    path = root / "cifar-10-binary.tar.gz"
    if path.exists():
        if checksum(path) != CIFAR_MD5:
            raise ValueError(f"CIFAR-10 校验失败，请检查或移走 {path}")
        return path
    print(f"Downloading {CIFAR_URL}", flush=True)
    with tempfile.NamedTemporaryFile(dir=root, delete=False) as output:
        temporary = Path(output.name)
    try:
        with urllib.request.urlopen(CIFAR_URL, timeout=120) as source, temporary.open("wb") as dest:
            shutil.copyfileobj(source, dest)
        if checksum(temporary) != CIFAR_MD5:
            raise ValueError("下载的 CIFAR-10 与官方 MD5 不一致")
        os.replace(temporary, path)
    finally:
        temporary.unlink(missing_ok=True)
    return path


class CIFARImages(Dataset):
    def __init__(self, path, training):
        names = [f"data_batch_{i}.bin" for i in range(1, 6)] if training else ["test_batch.bin"]
        batches = {}
        with tarfile.open(path, "r|gz") as archive:
            for member in archive:
                name = Path(member.name).name
                if name in names and member.isfile():
                    records = np.frombuffer(archive.extractfile(member).read(), np.uint8)
                    batches[name] = records.reshape(10000, 3073)[:, 1:]
        self.images = np.concatenate([batches[name] for name in names]).reshape(-1, 3, 32, 32)

    def __len__(self):
        return len(self.images)

    def __getitem__(self, index):
        return torch.from_numpy(self.images[index]).float().div_(255)


class FolderImages(Dataset):
    """从划分目录递归读图。size>0 等比例缩放并中心裁剪，size=0 保持原始尺寸。"""
    def __init__(self, root, size):
        extensions = {".jpg", ".jpeg", ".png", ".bmp", ".tif", ".tiff", ".webp"}
        self.files = sorted(p for p in root.rglob("*") if p.suffix.lower() in extensions and p.is_file())
        if not self.files:
            raise ValueError(f"未找到图片：{root}")
        self.size = size

    def __len__(self):
        return len(self.files)

    def __getitem__(self, index):
        with Image.open(self.files[index]) as image:
            rgb = image.convert("RGB")
            if self.size:
                rgb = ImageOps.fit(rgb, (self.size, self.size), method=Image.Resampling.BICUBIC)
            return torch.from_numpy(np.array(rgb)).permute(2, 0, 1).float().div_(255)


class SyntheticImages(Dataset):
    """仅作离线冒烟测试，每个样本独立生成低频色块，不参与 CIFAR-10 结论。"""
    def __init__(self, count, seed):
        self.count, self.seed = count, seed

    def __len__(self):
        return self.count

    def __getitem__(self, index):
        gen = torch.Generator().manual_seed(self.seed + index)
        small = torch.rand(1, 3, 5, 5, generator=gen)
        return torch.nn.functional.interpolate(small, size=(32, 32), mode="bilinear",
                                               align_corners=False)[0]


def datasets(cfg, training):
    """返回训练/验证二元组，或单独的测试集；先固定划分，再按 limit 截断。"""
    root = Path(cfg.data_dir)
    splits = ("train", "val") if training else ("test",)
    if cfg.dataset == "cifar10":
        images = CIFARImages(cifar_archive(root), training)
        if training:
            order = torch.randperm(len(images), generator=torch.Generator().manual_seed(cfg.seed))
            k = cfg.validation_holdout
            groups = [Subset(images, order[k:].tolist()), Subset(images, order[:k].tolist())]
        else:
            groups = [images]
    elif cfg.dataset == "folder":
        groups = [FolderImages(root / split, cfg.image_size) for split in splits]
    else:
        groups = [SyntheticImages(128, cfg.seed + 10000 * ("train", "val", "test").index(split))
                  for split in splits]
    limits = (cfg.train_limit, cfg.val_limit) if training else (cfg.test_limit,)
    result = []
    for group, limit, split in zip(groups, limits, splits):
        if limit and limit < len(group):
            gen = torch.Generator().manual_seed(cfg.seed + ("train", "val", "test").index(split))
            group = Subset(group, torch.randperm(len(group), generator=gen)[:limit].tolist())
        result.append(group)
    return tuple(result) if training else result[0]
```

## metrics.py

```python
"""逐图 RGB PSNR、局部 Gaussian SSIM，以及活动物理分支误码计数。"""
import torch
import torch.nn.functional as F


def image_scores(original, reconstructed):
    """返回各图 PSNR(dB) 与 SSIM。动态范围 1，完全相同的图像 PSNR 为正无穷。"""
    mse = (original - reconstructed).square().mean(dim=(1, 2, 3))
    psnr = -10 * torch.log10(mse)
    window = min(11, *original.shape[-2:])
    window -= 1 - window % 2
    coords = torch.arange(window, device=original.device, dtype=original.dtype) - window // 2
    gaussian = torch.softmax(-coords.square() / (2 * 1.5 ** 2), dim=0)
    kernel = torch.outer(gaussian, gaussian).expand(3, 1, window, window).contiguous()
    fields = torch.cat((original, reconstructed, original.square(), reconstructed.square(),
                        original * reconstructed))
    mx, my, ex2, ey2, exy = F.conv2d(fields, kernel, groups=3).chunk(5)
    luminance = (2 * mx * my + 0.01 ** 2) / (mx.square() + my.square() + 0.01 ** 2)
    structure = (2 * (exy - mx * my) + 0.03 ** 2) / (
        (ex2 - mx.square()).clamp_min(0) + (ey2 - my.square()).clamp_min(0) + 0.03 ** 2)
    return psnr, (luminance * structure).mean(dim=(1, 2, 3))


def bit_counts(sent, received):
    """每色原始错误数和比较位数；关闭颜色两项均记 0，BER 留空。"""
    active = (received.weights > 0).long()
    errors = (sent != received.bits).sum(dim=(0, 2)) * active
    totals = active * (sent.shape[0] * sent.shape[2])
    return errors.tolist(), totals.tolist()
```

```python
"""训练两组独立模型。两组复用同一初始化、训练样本顺序、嵌套掩码和优化目标。"""
import os
import tempfile
import uuid
from dataclasses import asdict

import numpy as np
import torch
import torch.nn.functional as F
from torch.utils.data import DataLoader

from channel.mcpam_channel import MCPAMChannel
from config import initialize, parameters
from data import datasets
from metrics import image_scores
from models.semantic_net import SemanticNet


@torch.inference_mode()
def validate(model, link, loader, cfg, device):
    """固定验证噪声，按图像和工作点等权平均 PSNR，单位 dB。"""
    model.eval()
    link.eval()
    generator = torch.Generator().manual_seed(cfg.seed + 91)
    total, count = 0., 0
    for images in loader:
        images = images.to(device)
        bits, latent = model.encode(images)
        for mode, snr in cfg.val_points:
            noise = torch.randn(len(images), link.slots_per_word, bits.size(2), generator=generator).to(device)
            received = link(bits, snr, mode, noise)
            decoded = model.decode(received.bits, latent, images.shape[-2:])
            psnr, _ = image_scores(images, decoded)
            total += psnr.sum().item()
            count += len(images)
    return total / count


def save_best(path, state):
    """同目录临时写入并原子替换权重，避免中断留下半个 checkpoint。"""
    with tempfile.NamedTemporaryFile(dir=path.parent, delete=False) as handle:
        temporary = handle.name
    try:
        torch.save(state, temporary)
        os.replace(temporary, path)
    finally:
        if os.path.exists(temporary):
            os.unlink(temporary)


def fit(cfg, name, ratios, train_set, val_set, experiment):
    device = initialize(cfg)
    rng = np.random.default_rng(cfg.seed)
    shuffle = torch.Generator().manual_seed(cfg.seed)
    batches = DataLoader(train_set, cfg.batch_size, shuffle=True, generator=shuffle)
    validation = DataLoader(val_set, cfg.batch_size)
    model = SemanticNet(cfg.channels, cfg.width).to(device)
    link = MCPAMChannel(ratios, cfg.compression, cfg.balanced_pairs).to(device)
    optimizer = torch.optim.AdamW(model.parameters(), lr=cfg.lr)
    path = cfg.directory("checkpoints") / f"{name}_best.pt"
    path.parent.mkdir(parents=True, exist_ok=True)
    best = -float("inf")
    for epoch in range(1, cfg.epochs + 1):
        model.train()
        link.train()
        loss_sum = 0.
        for images in batches:
            images = images.to(device)
            # 每个 minibatch 抽取一个前缀。低阶模式完整屏蔽后缀，且与测试时功率重分配一致。
            mode = int(rng.choice((2, 4, 8), p=cfg.mode_probs))
            snr = float(rng.uniform(*cfg.train_snr))
            prediction, _, _ = model(images, link, snr, mode)
            loss = F.mse_loss(prediction, images)
            optimizer.zero_grad(set_to_none=True)
            loss.backward()
            optimizer.step()
            loss_sum += loss.detach().item() * len(images)
        score = validate(model, link, validation, cfg, device)
        print(f"{name} {epoch:02}/{cfg.epochs} MSE={loss_sum / len(train_set):.5f} "
              f"val PSNR={score:.3f} dB", flush=True)
        if score > best:
            best = score
            save_best(path, dict(model=model.state_dict(), config=asdict(cfg), ratios=ratios,
                                 role=name, epoch=epoch, val_psnr=score, experiment=experiment,
                                 train_images=len(train_set), val_images=len(val_set)))


def main():
    cfg = parameters(__doc__, training=True)
    device = initialize(cfg)
    train_set, val_set = datasets(cfg, training=True)
    print(f"{device} | {cfg.dataset} | train={len(train_set)} val={len(val_set)}", flush=True)
    experiment = uuid.uuid4().hex
    for name, ratios in (("uniform", cfg.uniform), ("proposed", cfg.proposed)):
        fit(cfg, name, ratios, train_set, val_set, experiment)


if __name__ == "__main__":
    main()
```

```python
"""配对评估四种方案。每张测试图共享符号预算、标准噪声样本和平均光功率。"""
import csv
import json
import platform
from dataclasses import asdict, dataclass, field
from pathlib import Path

import matplotlib
import numpy as np
import PIL
import torch
from torch.utils.data import DataLoader

from baselines.jpeg_mcpam import pack, unpack
from channel.mcpam_channel import MCPAMChannel
from config import COLORS, SCHEMES, initialize, parameters
from data import datasets
from metrics import bit_counts, image_scores
from models.semantic_net import SemanticNet
from plots import save_ber, save_examples, save_quality


def counts_as_dict(errors, totals):
    fields = {}
    for i, color in enumerate(COLORS):
        fields[f"errors_{color}"] = int(errors[i])
        fields[f"bits_{color}"] = int(totals[i])
        fields[f"ber_{color}"] = errors[i] / totals[i] if totals[i] else None
    return fields


@dataclass
class Accumulator:
    images: int = 0
    slots: int = 0
    psnr: float = 0.
    ssim: float = 0.
    optical: float = 0.
    errors: np.ndarray = field(default_factory=lambda: np.zeros(3, dtype=np.int64))
    totals: np.ndarray = field(default_factory=lambda: np.zeros(3, dtype=np.int64))
    jpeg_failures: int = 0
    jpeg_overflows: int = 0
    jpeg_bytes: int = 0

    def add(self, original, decoded, bits, received, slots_per_word):
        psnr, ssim = image_scores(original, decoded)
        errors, totals = bit_counts(bits, received)
        self.errors += errors
        self.totals += totals
        self.psnr += psnr.sum().item()
        self.ssim += ssim.sum().item()
        self.images += len(original)
        slots_per_image = slots_per_word * bits.size(2)
        self.slots += len(original) * slots_per_image
        self.optical += received.optical_mean.sum().item() * slots_per_image

    def row(self, scheme, snr, mode):
        jpeg = scheme == SCHEMES[0]
        return dict(scheme=scheme, snr_db=snr, mode=mode, images=self.images,
                    physical_slots=self.slots, mean_optical_power=self.optical / self.slots,
                    psnr=self.psnr / self.images, ssim=self.ssim / self.images,
                    jpeg_decode_failures=self.jpeg_failures if jpeg else None,
                    jpeg_budget_failures=self.jpeg_overflows if jpeg else None,
                    jpeg_mean_bytes=self.jpeg_bytes / self.images if jpeg else None,
                    **counts_as_dict(self.errors, self.totals))


def load_models(cfg, device):
    models, links, metadata = {}, {}, {}
    for name, ratios in (("uniform", cfg.uniform), ("proposed", cfg.proposed)):
        path = cfg.directory("checkpoints") / f"{name}_best.pt"
        if not path.exists():
            raise FileNotFoundError(f"找不到 {path}，请先以同一 --run 执行 train.py")
        state = torch.load(path, map_location="cpu", weights_only=True)
        train_cfg = state["config"]
        if (tuple(state["ratios"]) != ratios or state["role"] != name or
                train_cfg.get("balanced_pairs", True) != cfg.balanced_pairs or
                any(train_cfg[k] != getattr(cfg, k) for k in ("channels", "width", "compression"))):
            raise ValueError(f"{path.name} 与当前网络或信道配置不一致")
        model = SemanticNet(cfg.channels, cfg.width)
        model.load_state_dict(state.pop("model"))
        models[name] = model.to(device).eval()
        links[name] = MCPAMChannel(ratios, cfg.compression, cfg.balanced_pairs).to(device).eval()
        metadata[name] = state
    if metadata["uniform"]["experiment"] != metadata["proposed"]["experiment"]:
        raise ValueError("两组权重来自不同训练运行，请完成同一次训练再进行对照")
    return models, links, metadata


def probe_ber(cfg, links, device):
    """使用同一组三路 IID 位流和噪声，独立比较权重本身的误码影响。"""
    gen = torch.Generator().manual_seed(cfg.seed + 23)
    shape = (1, 3, cfg.probe_symbols)
    bits = torch.randint(2, shape, generator=gen).float().to(device)
    noise = torch.randn(1, links["uniform"].slots_per_word, cfg.probe_symbols, generator=gen).to(device)
    rows = []
    for snr in cfg.snrs:
        for name, scheme in zip(("uniform", "proposed"), SCHEMES[1:3]):
            received = links[name](bits, snr, 8, noise)
            rows.append(dict(snr_db=snr, scheme=scheme, **counts_as_dict(*bit_counts(bits, received))))
    return rows


@torch.inference_mode()
def evaluate(cfg):
    device = initialize(cfg)
    models, links, metadata = load_models(cfg, device)
    test = datasets(cfg, training=False)
    loader = DataLoader(test, cfg.batch_size)
    gen = torch.Generator().manual_seed(cfg.seed + 17)
    accum = {(s, db): Accumulator() for s in SCHEMES for db in cfg.snrs}
    examples = {}
    for batch, images in enumerate(loader):
        images = images.to(device)
        encoded = {key: model.encode(images) for key, model in models.items()}
        slots = encoded["proposed"][0].size(2)
        frames = pack(images, slots, cfg.jpeg_qualities)
        slots_per_word = links["uniform"].slots_per_word
        noise = torch.randn(len(images), slots_per_word, slots, generator=gen).to(device)
        for db in cfg.snrs:
            capture = batch == 0 and db in (5, 15)
            if capture:
                examples[db] = [images[0].cpu()]
            for index, scheme in enumerate(SCHEMES):
                name = "uniform" if index < 2 else "proposed"
                mode = cfg.mode(db) if index == 3 else 8
                bits, latent = encoded[name]
                acc = accum[scheme, db]
                if index == 0:
                    bits = frames.bits
                received = links[name](bits, db, mode, noise)
                if index == 0:
                    decoded, failures = unpack(received.bits, images.shape[-2:])
                    acc.jpeg_failures += failures
                    acc.jpeg_overflows += frames.sizes.count(0)
                    acc.jpeg_bytes += sum(frames.sizes)
                else:
                    decoded = models[name].decode(received.bits, latent, images.shape[-2:])
                acc.add(images, decoded, bits, received, slots_per_word)
                if capture:
                    examples[db].append(decoded[0].cpu())
    rows = [a.row(s, db, cfg.mode(db) if s == SCHEMES[3] else 8) for (s, db), a in accum.items()]
    if len({r["physical_slots"] for r in rows}) != 1 or (cfg.balanced_pairs and any(
            abs(r["mean_optical_power"] - 0.5) > 1e-6 for r in rows)):
        raise RuntimeError("各方案的符号数或平均光功率核对失败")
    probe = probe_ber(cfg, links, device)
    directory = Path(cfg.output_dir) if cfg.output_dir else cfg.directory("results")
    directory.mkdir(parents=True, exist_ok=True)
    with (directory / "metrics.csv").open("w", newline="", encoding="utf-8-sig") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)
    record = dict(config=asdict(cfg), checkpoints=metadata, device=str(device), python=platform.python_version(),
                  torch=str(torch.__version__), numpy=np.__version__, pillow=PIL.__version__,
                  matplotlib=matplotlib.__version__, cuda=torch.version.cuda,
                  test_images=len(test), iid_ber=probe)
    (directory / "run.json").write_text(json.dumps(record, indent=2, ensure_ascii=False), encoding="utf-8")
    save_quality(rows, directory, cfg)
    save_ber(rows, probe, directory, cfg)
    save_examples(examples, directory)
    print(f"{len(test)} images | {len(rows)} points | {directory}", flush=True)


if __name__ == "__main__":
    evaluate(parameters(__doc__, training=False))
```

```python
"""针对信道标定、功率/时隙预算、STE、JPEG 和指标的可重复回归测试。"""
import itertools
import unittest

import torch

from baselines.jpeg_mcpam import pack, unpack
from channel.mcpam_channel import MCPAMChannel
from config import Config
from metrics import bit_counts, image_scores
from models.semantic_net import SemanticNet


class SimulationTests(unittest.TestCase):
    def setUp(self):
        torch.manual_seed(15)
        torch.set_num_threads(2)

    def test_noiseless_all_words_and_modes(self):
        bits = torch.tensor(list(itertools.product((0., 1.), repeat=3))).unsqueeze(-1)
        for k, ratio, mode in itertools.product((0., 0.8, 1.8), ((4, 2, 1), (6, 2, 1)), (2, 4, 8)):
            link = MCPAMChannel(ratio, k).eval()
            rx = link(bits, 10, mode, torch.zeros(8, 2, 1))
            mask = (rx.weights > 0).view(1, 3, 1)
            torch.testing.assert_close(rx.bits, bits * mask)
            torch.testing.assert_close(rx.optical_mean, torch.full((8,), 0.5))
            self.assertAlmostEqual(rx.weights.sum().item(), 1., places=6)
            errors, totals = bit_counts(bits, rx)
            self.assertEqual(errors, [0, 0, 0])
            self.assertEqual(totals, [8 if w > 0 else 0 for w in rx.weights])

    def test_sigma_and_ook_theory(self):
        # OOK 互补合并后的误码率是 Q(sqrt(2)*10^(SNR/20))，用它端到端检查噪声定标。
        import math
        link = MCPAMChannel().eval()
        for db in (0, 5, 25):
            self.assertAlmostEqual(link.noise_sigma(db) ** 2, .25 * 10 ** (-db / 10))
        bits = torch.randint(2, (1, 3, 150000)).float()
        rx = link(bits, 0, 2)
        errors, totals = bit_counts(bits, rx)
        self.assertAlmostEqual(errors[0] / totals[0], math.erfc(1.) / 2, delta=.003)

    def test_raw_single_slot_levels(self):
        bits = torch.tensor(list(itertools.product((0., 1.), repeat=3))).unsqueeze(-1)
        link = MCPAMChannel((4, 2, 1), compression=0, balanced_pairs=False).eval()
        rx = link(bits, 0, noise=torch.zeros(8, 1, 1))
        torch.testing.assert_close(rx.optical_mean, torch.arange(8) / 7.)
        torch.testing.assert_close(rx.bits, bits)

    def test_ste_and_inactive_gradients(self):
        bits = torch.ones(1, 3, 8, requires_grad=True)
        rx = MCPAMChannel().train()(bits, 0, 4)
        rx.bits.sum().backward()
        torch.testing.assert_close(bits.grad[:, :2], torch.ones_like(bits[:, :2]))
        self.assertEqual(bits.grad[:, 2].count_nonzero().item(), 0)
        net = SemanticNet(4, 8)
        x = torch.rand(2, 3, 31, 35)
        y, sent, _ = net(x, MCPAMChannel(), 5, 2)
        self.assertEqual(y.shape, x.shape)
        self.assertEqual(sent.size(-1), 4 * 16 * 18)
        (y - x).square().mean().backward()
        grad = net.encoder[0].weight.grad
        self.assertTrue(torch.isfinite(grad).all() and grad.abs().sum() > 0)

    def test_jpeg_budget_and_failure(self):
        x = torch.rand(2, 3, 32, 32)
        frames = pack(x, 2048, Config().jpeg_qualities)
        self.assertEqual(frames.bits.shape, (2, 3, 2048))
        self.assertTrue(all(0 < size <= 768 for size in frames.sizes))
        y, failures = unpack(frames.bits, (32, 32))
        self.assertEqual(failures, 0)
        self.assertEqual(y.shape, x.shape)
        tiny = pack(x, 32, Config().jpeg_qualities)
        self.assertEqual(tiny.sizes, [0, 0])
        grey, failures = unpack(tiny.bits, (32, 32))
        self.assertEqual(failures, 2)
        torch.testing.assert_close(grey, torch.full_like(x, .5))

    def test_ssim_and_thresholds(self):
        x = torch.rand(2, 3, 32, 32)
        psnr, ssim = image_scores(x, x.clone())
        self.assertTrue(torch.isposinf(psnr).all())
        torch.testing.assert_close(ssim, torch.ones(2))
        zeros, ones = torch.zeros_like(x), torch.ones_like(x)
        psnr, ssim = image_scores(zeros, ones)
        torch.testing.assert_close(psnr, torch.zeros(2))
        torch.testing.assert_close(ssim, torch.full((2,), .0001 / 1.0001))
        self.assertEqual([Config().mode(s) for s in (0, 7.49, 7.5, 17.49, 17.5, 25)],
                         [2, 2, 4, 4, 8, 8])


if __name__ == "__main__":
    unittest.main()
```

````markdown
## 单 PD 信道与公平性

1. 每个信息字包含 R、B、G 三个分支位。先关闭当前模式停用的分支，再对活动权重归一化。设活动分支权重为 $w_i$，线性驱动为 $x=\sum_i w_i b_i$，峰值为 1。线性 8 阶模式的 4∶2∶1 产生等距的 0 到 1 八个电平。6∶2∶1 增大红路两个子星座之间的间隔。
2. 静态非线性使用公共驱动与光源链路的等效标定曲线 $f(x)=0.5+0.5\tanh(k(2x-1))/\tanh(k)$。默认 $k=0.8$，设为 0 可得到线性信道。它满足 $f(1-x)=1-f(x)$。模型假设三色接收增益已校准到同一个单位，PD 响应度取 1。具体 LED 的分色响应、带宽、散粒噪声以及接收机标定误差需结合硬件数据另行建模。
3. 默认对每个信息字发送 $x$、$1-x$ 两个物理时隙。第二时隙等价于翻转全部活动位，因此任意位流的时隙平均发射光强都是 0.5，峰值预算都是 1。各模式共享相同时隙数。两次 PD 采样独立加噪后，通过 $(y_1+1-y_2)/2$ 合并，合并噪声方差为单时隙的一半。两次采样均计入传输开销。
4. 横轴采用各方案共用的参考电 SNR，$\mathrm{SNR}_{ref}=10\log_{10}((0.5)^2/\sigma^2)$。每个物理时隙的 AWGN 方差为 $0.25\,10^{-\mathrm{SNR}_{dB}/10}$，数据是一组有限高斯样本。该定义维持同一噪声底，星座的交流方差会随权重改变。需要原始单时隙叠加时将 `balanced_pairs=False`，随后重新训练和评估；此时只约束同峰值，实际平均光强写入 CSV，横轴仍以 0.5 为参考。
5. 接收机按 R、B、G 逐层判决。令已判前缀为 $p$，当前权重为 $w$，剩余权重和为 $t$，用 $f(p+t)$ 与 $f(p+w)$ 的中点分开两个子星座。判完一位，从残差中扣除 $f(p+w\hat b)-f(p)$。当 $k=0$ 时，扣除量恰好为 $w\hat b$。权重需满足 $w_R>w_B+w_G$ 及 $w_B>w_G>0$，才能使用这里的前缀硬判决。
6. 8 阶驱动三色，4 阶使用红蓝，2 阶只使用红光。默认阈值为 7.5 和 17.5 dB，恰在阈值时切到较高阶。发端和收端假设共享模式及 SNR 估计，模型采用理想同步。阈值是可调的实验参数。

默认 32×32 图像经过步长 2 的编码器，三路各有 $8\times16\times16=2048$ 位。每图占用 **4096 个物理时隙**。JPEG 最多使用 6144 位，即 **768 字节**。2、4、8 阶每图携带的信息位数分别为 2048、4096、6144。

## 语义分层与训练

编码器先降采样，再用残差块生成连续的 24 个特征通道。按固定位置分成三组，每组 8 个。量化前向采用硬阈值，反向用 sigmoid 近似梯度。信道同样执行真实硬判决，反向把活动接收位的梯度直通到对应发射位。停用分支保持零值和零梯度。解码器用卷积与 PixelShuffle 上采样到 RGB 图像，全程只接收解调特征。

每个训练批次在 0～25 dB 连续均匀采样 SNR，并以 0.25、0.25、0.50 的概率只保留 R、保留 R+B 或保留全部。掩码始终是前缀，对应 Nested Dropout。每种模式都以原图为目标优化 MSE，红路因此在所有批次中都承担重建任务，蓝绿路逐步获得提供补充信息的机会。轮廓、主体、高频的语义分工需要从重建图与后续任务指标验证。PSNR 和 SSIM 直接衡量图像重建保真度。

两套模型从相同初始参数训练，复用数据顺序、SNR、噪声随机流、掩码和优化目标。唯一主动改变的实验变量为光强权重。每轮用固定验证噪声，在 `config.val_points` 的四个模式/SNR 组合上计算逐图平均 PSNR，保存得分最高的权重。固定和动态 Proposed 共享同一最佳权重。两套模型保存相同的训练批次标识，评估入口会检查兼容性。

## 对照实验与读图

| 曲线 | 源编码 | 活动模式 | 光强比例（R∶B∶G） |
|---|---|---|---|
| JPEG 4:2:1 | 完整 JPEG 比特流 | 固定 8 阶 | 4∶2∶1 |
| Semantic uniform | 分层 CNN，独立训练 | 固定 8 阶 | 4∶2∶1 |
| Hierarchical 8-CPAM | 分层 CNN | 固定 8 阶 | 6∶2∶1 |
| Hierarchical adaptive | 上一行的同一 CNN | 按 SNR 切换 | 活动颜色重新归一 |

JPEG 使用 Pillow 编码，色度抽样 4∶2∶0。按质量候选从高到低试编码，整个文件含 JPEG 头、量化表和结束标记都必须装入本图预算。剩余位补零，再轮流映射到 R、B、G。解码器直接读取接收端固定帧，事先只知道图像尺寸。超预算或码流损坏到无法解码时输出中灰图，仍纳入平均指标。可解码的损坏图保留其实际像素。CSV 分别保存预算不足和解码失败的计数，前者可能同时计入后者。该基线采用无 FEC、无重传的 JPEG 链路。

- `metrics.csv` 有四组方案 × 六个 SNR 点，共 24 行。PSNR 按逐图 dB 平均；SSIM 在 RGB 各通道使用 11×11、标准差 1.5 的高斯窗，取有效区域及三通道均值，像素范围 [0,1]。小图用可容纳的最大奇数窗。每色 BER 是总错误数除以总比较位数，停用色分母为零，BER 留空。JPEG 的 BER 含实际发送的填充位。
- `physical_slots` 是该行所有图的累计时隙数，除以 `images` 可得逐图平均。`mean_optical_power` 由各帧实际发射波形按时隙数加权统计。默认模式下，评估结束前核对四组的物理时隙数和平均光功率。
- `psnr_vs_snr.png` 与 `ssim_vs_snr.png` 使用实际评估值。高 SNR 的动态方案和固定 8 阶使用同一模型、同一模式和同一噪声，两条曲线应当重合。
- `ber_vs_snr.png` 左侧是实际语义位流，右侧是同一组三路独立等概率随机位和同一组噪声下的对照，可区分内容分布与星座权重的影响。零误码点在对数轴以空心倒三角标记，显示位置为 $0.5/N$，其原始 BER 仍是零。随机探针的原始分子、分母在 `run.json`。
- `visual_comparison.png` 使用固定测试顺序的第一张图，两行分别是 5 和 15 dB，五列是原图与四种重建。两行和所有方案共用原图。
- `run.json` 还记录训练和测试配置、环境、最佳轮次与实际样本数。结论应注明种子、样本规模、权重、阈值、非线性和 SNR 定义。不同后端或依赖版本可能出现浮点差异。

## 随项目交付的 CIFAR-10 仿真结果

本次用 CPU 版 PyTorch 2.9.1、种子 2026 运行。两组各训练 8 轮，均选中第 8 轮。每个测试点统计 10,000 张图，累计 40,960,000 个物理时隙，各组实算平均光强均为 0.5。表中每格依次是平均 PSNR（dB）和 SSIM。

| SNR（dB） | JPEG 4∶2∶1 | 语义 4∶2∶1 | 分级 6∶2∶1 固定 8 阶 | 分级 6∶2∶1 动态 |
|---:|---:|---:|---:|---:|
| 0  | 12.51 / 0.130 | 23.99 / 0.834 | 22.92 / 0.816 | 23.39 / 0.821 |
| 5  | 12.51 / 0.130 | 26.37 / 0.896 | 25.08 / 0.883 | 25.99 / 0.894 |
| 10 | 12.51 / 0.130 | 27.39 / 0.915 | 26.14 / 0.905 | 26.55 / 0.905 |
| 15 | 12.51 / 0.130 | 27.88 / 0.923 | 26.63 / 0.914 | 27.09 / 0.915 |
| 20 | 12.51 / 0.130 | 28.14 / 0.926 | 26.86 / 0.917 | 26.86 / 0.917 |
| 25 | 17.39 / 0.409 | 28.22 / 0.928 | 26.98 / 0.919 | 26.98 / 0.919 |

15 dB 时，将固定 8 阶方案从 4∶2∶1 改为 6∶2∶1，真实语义位流的红路 BER 从 0.961% 降至 0.0139%，绿路 BER 从 20.28% 升至 25.46%。此次固定 6∶2∶1 在六个 SNR 点的 PSNR 和 SSIM 均低于均匀权重。动态降阶在 5 dB 把 6∶2∶1 的 PSNR 提高约 0.90 dB，在 15 dB 提高约 0.47 dB。单个种子的这一组结果支持“增强红路保护”，尚不足以支持“整体图像重建更优”。

JPEG 在 0 至 20 dB 的每个测试点均有 10,000 帧解码失败，曲线对应规定的中灰补偿结果。25 dB 时有 5,315 帧解码失败。所有测试图都能在发送端找到预算内的 JPEG 质量。对该基线的解释取决于有无纠错、报头保护和失败补偿策略。

## 工程问题与验证

| 问题 | 处理方式 | 验证 |
|---|---|---|
| 码流“1”的比例不同导致平均光功率漂移 | 原码和反码按物理时隙配对 | 枚举所有输入字逐帧检查平均光强 |
| LED 非线性改变判决边界 | 从标定曲线计算局部阈值和剥离量 | 线性和两种非线性强度均检查三种模式 |
| 硬二值化和硬判决截断梯度 | 编码量化 STE、按活动分支直通 | 检查编码器梯度非零和停用分支梯度为零 |
| JPEG 文件头或码流装不进帧 | 自适应质量、计入头部、失败帧中灰重建 | 检查预算内无噪声解码、极小预算失败行为 |
| 误把关闭通道或有限样本零误码读成高可靠性结论 | 输出实际错误数和比较位数 | 关闭分支留空，零误码用单独图标 |
| 单时隙和合并后噪声方差可能混淆 | 从共同参考功率推导每时隙噪声 | 0 dB 下 OOK 仿真与解析值 $Q(\sqrt 2)$ 对照 |
````
