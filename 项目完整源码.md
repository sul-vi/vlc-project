````python
"""实验参数、命令行入口与随机性控制。路径均相对于项目目录。"""
import argparse
import os
import random
from dataclasses import dataclass
from pathlib import Path

import numpy as np
import torch

ROOT = Path(__file__).resolve().parent
COLORS = ("R", "B", "G")
JPEG, UNORDERED, ORDERED, FIXED, ADAPTIVE = (
    "JPEG + FEC 4:2:1", "Unordered semantic 4:2:1", "Ordered semantic 4:2:1",
    "Proposed 6:2:1 fixed", "Proposed 6:2:1 adaptive")
SCHEMES = (JPEG, UNORDERED, ORDERED, FIXED, ADAPTIVE)
RAW_JPEG = "JPEG uncoded 4:2:1"


@dataclass(frozen=True)
class Config:
    dataset: str = "cifar10"
    data_dir: str = str(ROOT / "data")
    image_size: int = 32       # 文件夹数据集的正方形裁剪边长，0 保持原分辨率
    uniform: tuple = (4., 2., 1.)
    proposed: tuple = (6., 2., 1.)
    channels: int = 16        # 每色 16×8×8 = 1024 位
    width: int = 32
    compression: float = 0.8  # 中点对称 tanh 的强度，0 对应线性响应
    balanced_pairs: bool = True  # True 保证逐帧平均光功率相同，False 运行原始单时隙 M-CPAM
    candidates: tuple = ((6., 2., 1.),)
    classifier_epochs: int = 14
    task_weight: float = 0.5
    snrs: tuple = (-5, 0, 5, 10, 15, 20, 25)
    train_snr: tuple = (-5., 25.)
    mode_probs: tuple = (0.25, 0.25, 0.50)  # 分别对应 2、4、8 阶
    val_points: tuple = ((2, 5.), (4, 10.), (8, 5.), (8, 20.))
    lr: float = 0.001
    epochs: int = 30
    batch_size: int = 64
    validation_holdout: int = 5000
    train_limit: int = 15000  # 0 表示全部，三个 limit 只截取各自的集合
    val_limit: int = 1024
    test_limit: int = 0
    seed: int = 2026
    device: str = "auto"
    run: str = "v3"
    output_dir: str = ""      # 可指定独立评估输出目录，方便同一权重评估不同数据集
    eval_ratio: str = ""      # 留空时使用验证集选出的比例，例如 6:2:1 可固定候选
    jpeg_qualities: tuple = (95, 90, 85, 80, 75, 70, 60, 50, 40, 30, 20, 10, 5, 1)
    probe_symbols: int = 262144  # BER 对照中的独立三比特字数

    def directory(self, name):
        return ROOT / name / self.run


def parameters(description, training):
    """读取公共参数与当前入口支持的覆盖项。"""
    parser = argparse.ArgumentParser(description=description)
    parser.add_argument("--dataset", choices=("cifar10", "folder", "synthetic"))
    parser.add_argument("--device", choices=("auto", "cpu", "mps", "cuda"))
    parser.add_argument("--run", help="checkpoints 和 results 下的实验名")
    parser.add_argument("--data-dir")
    if not training:
        parser.add_argument("--output-dir", help="评估输出位置，默认 results/<run>")
        parser.add_argument("--eval-ratio", help="固定一个受训候选，例如 5:2:1 或 6:2:1")
    keys = ["batch_size", "seed", "image_size"]
    keys += ["epochs", "train_limit", "val_limit"] if training else ["test_limit"]
    for key in keys:
        parser.add_argument("--" + key.replace("_", "-"), type=int)
    args = {k: v for k, v in vars(parser.parse_args()).items() if v is not None}
    cfg = Config(**args)
    if cfg.run and (Path(cfg.run).name != cfg.run or cfg.run in (".", "..")):
        parser.error("--run 必须是单个目录名")
    if min(cfg.batch_size, cfg.epochs, cfg.channels, cfg.width) < 1:
        parser.error("batch size、轮数和网络宽度必须为正数")
    if min(cfg.train_limit, cfg.val_limit, cfg.test_limit, cfg.image_size, cfg.seed) < 0:
        parser.error("样本上限、图像尺寸和种子须为非负数")
    if cfg.dataset == "folder" and cfg.image_size == 0 and cfg.batch_size != 1:
        parser.error("文件夹原尺寸模式请使用 --batch-size 1，以支持不同尺寸")
    return cfg


def initialize(cfg):
    """设置随机源和设备。相同软件、设备和参数下可复现实验。"""
    os.environ.setdefault("CUBLAS_WORKSPACE_CONFIG", ":4096:8")
    random.seed(cfg.seed)
    np.random.seed(cfg.seed)
    torch.manual_seed(cfg.seed)
    torch.backends.cudnn.benchmark = False
    torch.backends.cudnn.deterministic = True
    torch.use_deterministic_algorithms(True, warn_only=True)
    device = cfg.device
    if device == "auto":
        device = "cuda" if torch.cuda.is_available() else "mps" if torch.backends.mps.is_available() else "cpu"
    if device == "cpu":
        torch.set_num_threads(min(8, torch.get_num_threads()))
    return torch.device(device)
````

````python
"""图像源与固定数据划分。CIFAR-10 直接读取官方二进制归档，无需反序列化 pickle。"""
import hashlib
import os
import shutil
import tarfile
import tempfile
import urllib.request
from pathlib import Path

import numpy as np
import torch
from PIL import Image, ImageOps
from torch.utils.data import Dataset, Subset

CIFAR_URL = "https://cave.cs.toronto.edu/kriz/cifar-10-binary.tar.gz"
CIFAR_MD5 = "c32a1d4ab5d03f1284b67883e8d87530"


def checksum(path):
    digest = hashlib.md5()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1 << 20), b""):
            digest.update(block)
    return digest.hexdigest()


def cifar_archive(root):
    """仅在官方 MD5 匹配后原子发布归档。中断下载不会变成有效数据集。"""
    root.mkdir(parents=True, exist_ok=True)
    path = root / "cifar-10-binary.tar.gz"
    if path.exists():
        if checksum(path) != CIFAR_MD5:
            raise ValueError(f"CIFAR-10 校验失败，请检查或移走 {path}")
        return path
    print(f"Downloading {CIFAR_URL}", flush=True)
    with tempfile.NamedTemporaryFile(dir=root, delete=False) as output:
        temporary = Path(output.name)
    try:
        with urllib.request.urlopen(CIFAR_URL, timeout=120) as source, temporary.open("wb") as dest:
            shutil.copyfileobj(source, dest)
        if checksum(temporary) != CIFAR_MD5:
            raise ValueError("下载的 CIFAR-10 与官方 MD5 不一致")
        os.replace(temporary, path)
    finally:
        temporary.unlink(missing_ok=True)
    return path


class CIFARImages(Dataset):
    def __init__(self, path, training, labeled=False):
        self.labeled = labeled
        names = [f"data_batch_{i}.bin" for i in range(1, 6)] if training else ["test_batch.bin"]
        batches = {}
        with tarfile.open(path, "r|gz") as archive:
            for member in archive:
                name = Path(member.name).name
                if name in names and member.isfile():
                    records = np.frombuffer(archive.extractfile(member).read(), np.uint8)
                    batches[name] = records.reshape(10000, 3073)
        records = np.concatenate([batches[name] for name in names])
        self.labels = records[:, 0]
        self.images = records[:, 1:].reshape(-1, 3, 32, 32)

    def __len__(self):
        return len(self.images)

    def __getitem__(self, index):
        image = torch.from_numpy(self.images[index]).float().div_(255)
        return (image, int(self.labels[index])) if self.labeled else image


class FolderImages(Dataset):
    """从划分目录递归读图。size>0 等比例缩放并中心裁剪，size=0 保持原始尺寸。"""
    def __init__(self, root, size):
        extensions = {".jpg", ".jpeg", ".png", ".bmp", ".tif", ".tiff", ".webp"}
        self.files = sorted(p for p in root.rglob("*") if p.suffix.lower() in extensions and p.is_file())
        if not self.files:
            raise ValueError(f"未找到图片：{root}")
        self.size = size

    def __len__(self):
        return len(self.files)

    def __getitem__(self, index):
        with Image.open(self.files[index]) as image:
            rgb = image.convert("RGB")
            if self.size:
                rgb = ImageOps.fit(rgb, (self.size, self.size), method=Image.Resampling.BICUBIC)
            return torch.from_numpy(np.array(rgb)).permute(2, 0, 1).float().div_(255)


class SyntheticImages(Dataset):
    """仅作离线冒烟测试，每个样本独立生成低频色块，不参与 CIFAR-10 结论。"""
    def __init__(self, count, seed):
        self.count, self.seed = count, seed

    def __len__(self):
        return self.count

    def __getitem__(self, index):
        gen = torch.Generator().manual_seed(self.seed + index)
        small = torch.rand(1, 3, 5, 5, generator=gen)
        return torch.nn.functional.interpolate(small, size=(32, 32), mode="bilinear",
                                               align_corners=False)[0]


def datasets(cfg, training, labeled=False):
    """返回训练/验证二元组，或单独的测试集；先固定划分，再按 limit 截断。"""
    root = Path(cfg.data_dir)
    splits = ("train", "val") if training else ("test",)
    if cfg.dataset == "cifar10":
        images = CIFARImages(cifar_archive(root), training, labeled)
        if training:
            order = torch.randperm(len(images), generator=torch.Generator().manual_seed(cfg.seed))
            k = cfg.validation_holdout
            groups = [Subset(images, order[k:].tolist()), Subset(images, order[:k].tolist())]
        else:
            groups = [images]
    elif cfg.dataset == "folder":
        groups = [FolderImages(root / split, cfg.image_size) for split in splits]
    else:
        groups = [SyntheticImages(128, cfg.seed + 10000 * ("train", "val", "test").index(split))
                  for split in splits]
    limits = (cfg.train_limit, cfg.val_limit) if training else (cfg.test_limit,)
    result = []
    for group, limit, split in zip(groups, limits, splits):
        if limit and limit < len(group):
            gen = torch.Generator().manual_seed(cfg.seed + ("train", "val", "test").index(split))
            group = Subset(group, torch.randperm(len(group), generator=gen)[:limit].tolist())
        result.append(group)
    return tuple(result) if training else result[0]
````

## channel/mcpam_channel.py

````python
"""RGB 无滤光片、等效响应度为 1 的单 PD 链路。分支轴固定按 R、B、G 排列。"""
import math
from dataclasses import dataclass

import torch
from torch import nn


@dataclass
class Reception:
    bits: torch.Tensor          # [N,3,S]，关闭分支输出 0
    weights: torch.Tensor       # 活动光功率比例，总和为 1
    optical_mean: torch.Tensor  # [N]，LED 模型后实际发送的平均光强


class MCPAMChannel(nn.Module):
    """每个信息字发送原码与反码两个时隙，保证每帧平均光功率严格一致。"""

    def __init__(self, ratios=(6., 2., 1.), compression=0.8, balanced_pairs=True):
        super().__init__()
        if len(ratios) != 3 or not all(math.isfinite(x) for x in ratios):
            raise ValueError("需要三个有限的 R/B/G 权重")
        r, b, g = ratios
        if not (r > b + g and b > g > 0):
            raise ValueError("逐层判决要求 R > B + G 且 B > G > 0")
        if not math.isfinite(compression) or compression < 0:
            raise ValueError("LED 压缩系数必须有限且非负")
        self.register_buffer("ratios", torch.tensor(ratios, dtype=torch.float32))
        self.compression = compression
        self.slots_per_word = 2 if balanced_pairs else 1

    def weights(self, mode):
        """将本阶活动颜色的峰值光功率之和归一到 1。"""
        if mode not in (2, 4, 8):
            raise ValueError("mode 取 2、4 或 8")
        weights = self.ratios.clone()
        weights[int(math.log2(mode)):] = 0
        return weights / weights.sum()

    def led(self, x):
        """等效公共驱动的中点对称软限幅，f(0)=0、f(1)=1、f(1-x)=1-f(x)。"""
        k = self.compression
        if k == 0:
            return x
        return 0.5 + torch.tanh(k * (2 * x - 1)) * (0.5 / math.tanh(k))

    @staticmethod
    def noise_sigma(snr_db):
        """以单位响应度和平均光强 0.5 为参考，返回单时隙电流噪声标准差。"""
        return 0.5 * 10.0 ** (-snr_db / 20)

    def sic(self, received, weights, levels):
        """在 PD 电流域比较相邻子星座边界，在发射幅度域累计已判分量。"""
        prefix = torch.zeros_like(received)
        residual = received
        base = self.led(prefix)
        decoded = []
        for index in range(levels):
            weight = weights[index]
            tail = weights[index + 1:].sum()
            # 当低位取全 1 或全 0 时，两簇相邻端点经 LED 映射后取中点。
            threshold = (self.led(prefix + tail) + self.led(prefix + weight)) * 0.5 - base
            bit = (residual >= threshold).to(received.dtype)
            decoded.append(bit)
            prefix = prefix + weight * bit
            # 线性时这一步就是 residual -= weight * bit；软限幅时扣除标定的光强差。
            next_base = self.led(prefix)
            residual = residual - (next_base - base)
            base = next_base
        decoded.extend(torch.zeros_like(received) for _ in range(3 - levels))
        return torch.stack(decoded, dim=1)

    def forward(self, bits, snr_db, mode=8, noise=None):
        """输入浮点二进制 [N,3,S]，snr_db 为标量。noise 可传标准正态 [N,时隙数,S]。"""
        if bits.ndim != 3 or bits.shape[1] != 3 or not bits.is_floating_point():
            raise ValueError("输入形状须为 [批量,3,每色比特数]，元素为浮点 0/1")
        weights = self.weights(mode)
        x = (bits * weights.view(1, 3, 1)).sum(dim=1)
        pair = self.slots_per_word == 2
        clean = self.led(torch.stack((x, 1 - x), dim=1) if pair else x.unsqueeze(1))
        if noise is None:
            noise = torch.randn_like(clean)
        if noise.shape != clean.shape:
            raise ValueError("噪声必须与实际发送时隙的形状一致")
        received = clean + self.noise_sigma(snr_db) * noise
        combined = (received[:, 0] + 1 - received[:, 1]) * 0.5 if pair else received[:, 0]
        hard = self.sic(combined, weights, int(math.log2(mode)))
        if self.training:
            # 按数据流作硬判决 STE。前向保留真实 SIC 错误，反向对活动位用恒等近似。
            # 直接微分离散 SIC 会给出零梯度；停用颜色不能接收梯度。
            surrogate = bits * (weights > 0).view(1, 3, 1)
            hard = surrogate + (hard - surrogate).detach()
        return Reception(hard, weights, clean.mean(dim=(1, 2)))
````

````python
"""共享伪随机交织。每个逻辑层分布到三种物理颜色的位数最多相差一位。"""
import torch


def color_indices(bits, generator):
    """逐帧打乱轮转次序，返回 [批量,物理颜色,字序号] 对应的逻辑颜色索引。"""
    n, _, words = bits.shape
    shifts = torch.rand(n, words, generator=generator).argsort(dim=1).to(bits.device)
    colors = torch.arange(3, device=bits.device).view(1, 3, 1)
    return (colors + shifts.unsqueeze(1)) % 3


def restore(physical, indices):
    """按发端的索引解交织；每个信息字内的三色索引恰好是一个排列。"""
    return torch.zeros_like(physical).scatter(1, indices, physical)
````

## channel/mode_policy.py

````python
"""仅用训练留出集标定非均匀权重与模式切换边界。"""
import torch
import torch.nn.functional as F
from torch.utils.data import DataLoader

from channel.mcpam_channel import MCPAMChannel
from data import datasets


def choose_policy(rows, snrs, metric="psnr"):
    """最大化指定验证指标，允许降阶和回切，平分时优先较低阶。"""
    scores = {(r["snr_db"], r["mode"]): r[metric] for r in rows}
    modes = (2, 4, 8)
    path = [max(modes, key=lambda m: scores[db, m]) for db in snrs]
    switches = []
    current = path[0]
    for lo, hi in zip(snrs, snrs[1:]):
        # 三条线段的全部交点把区间切成若干段，每段内的优劣排序不变。
        cuts = {0., 1.}
        for i, a in enumerate(modes):
            for b in modes[i+1:]:
                left = scores[lo, a] - scores[lo, b]
                right = scores[hi, a] - scores[hi, b]
                if left * right < 0:
                    cuts.add(left / (left - right))
        knots = sorted(cuts)
        for start, end in zip(knots, knots[1:]):
            t = (start + end) / 2
            best = max(modes, key=lambda m: (1-t)*scores[lo, m] + t*scores[hi, m])
            if best != current:
                switches.append(dict(threshold_db=lo + start*(hi-lo), mode=best))
                current = best
    return dict(initial_mode=path[0], switches=switches, grid=list(zip(snrs, path)),
                metric=metric, mean_score=sum(scores[db, m] for db, m in zip(snrs, path))/len(path))


def active_mode(policy, db):
    for sample, selected in policy["grid"]:
        if db == sample:
            return selected
    mode = policy["initial_mode"]
    for switch in policy["switches"]:
        if db >= switch["threshold_db"]:
            mode = switch["mode"]
    return mode


@torch.inference_mode()
def calibrate(model, cfg, device):
    """有标签时按负联合损失选模式，同时保存 v2 的 PSNR 策略供对照。"""
    labeled = cfg.dataset == "cifar10"
    _, validation = datasets(cfg, True, labeled=labeled)
    links = [MCPAMChannel(r, cfg.compression, cfg.balanced_pairs).eval().to(device)
             for r in cfg.candidates]
    stats = {(i, db, mode): [0., 0, 0.] for i in range(len(links))
             for db in cfg.snrs for mode in (2, 4, 8)}
    gen = torch.Generator().manual_seed(cfg.seed + 73)
    for batch in DataLoader(validation, cfg.batch_size):
        x, labels = batch if labeled else (batch, None)
        x = x.to(device)
        if labeled:
            labels = labels.to(device)
        bits, shape = model.encode(x)
        noise = torch.randn(len(x), links[0].slots_per_word, bits.size(2), generator=gen).to(device)
        for (i, db, mode), totals in stats.items():
            rx = links[i](bits, db, mode, noise)
            y = model.decode(rx.bits, shape, x.shape[-2:], mode, db)
            totals[0] += (-10 * (x-y).square().mean((1,2,3)).log10()).sum().item()
            if labeled:
                logits = model.classify(rx.bits, shape)
                totals[1] += (logits.argmax(1) == labels).sum().item()
                cost = F.mse_loss(y, x) + cfg.task_weight * F.cross_entropy(logits, labels)
                totals[2] -= cost.item() * len(x)
    reports = []
    for i, ratio in enumerate(cfg.candidates):
        rows = [dict(snr_db=db, mode=mode, psnr=v[0]/len(validation),
                     acc=v[1]/len(validation) if labeled else None,
                     neg_dual_loss=v[2]/len(validation) if labeled else None)
                for (idx, db, mode), v in stats.items() if idx == i]
        reports.append(dict(ratio=ratio, rows=rows, psnr_policy=choose_policy(rows, cfg.snrs),
                            policy=choose_policy(rows, cfg.snrs, "neg_dual_loss" if labeled else "psnr")))
    return reports, max(reports, key=lambda r: r["policy"]["mean_score"])
````

````python
"""全卷积语义编解码器。通信分支 R/B/G 与输入图像 RGB 像素轴互相独立。"""
import torch
import torch.nn.functional as F
from torch import nn
from models.classifier import Classifier


class ResidualBlock(nn.Module):
    def __init__(self, width):
        super().__init__()
        self.body = nn.Sequential(
            nn.Conv2d(width, width, 3, padding=1),
            nn.GroupNorm(1, width),
            nn.SiLU(),
            nn.Conv2d(width, width, 3, padding=1),
        )

    def forward(self, x):
        return F.silu(x + self.body(x))


class SemanticNet(nn.Module):
    """每色占 C×ceil(H/4)×ceil(W/4) 个信息位，解码后裁回原尺寸。"""

    def __init__(self, channels=16, width=32):
        super().__init__()
        self.channels = channels
        self.encoder = nn.Sequential(
            nn.Conv2d(3, width, 5, stride=2, padding=2),
            nn.SiLU(), nn.Conv2d(width, width, 3, stride=2, padding=1),
            nn.SiLU(), ResidualBlock(width),
            nn.Conv2d(width, 2 * channels, 1),
        )
        # R 使用干净分类器的卷积结构，训练入口用同一训练划分的权重初始化。
        self.core_encoder = Classifier().features[:-1]
        self.core_bits = nn.Conv2d(96, channels, 1)
        self.task_head = nn.Sequential(
            nn.Conv2d(channels, 96, 3, padding=1), nn.BatchNorm2d(96), nn.ReLU(),
            nn.AdaptiveAvgPool2d((2, 2)), nn.Flatten(), nn.Linear(384, 10))
        self.decoder = nn.Sequential(
            nn.Conv2d(3 * channels + 4, width, 3, padding=1),
            nn.SiLU(), ResidualBlock(width),
            nn.Conv2d(width, 4 * width, 3, padding=1),
            nn.PixelShuffle(2), nn.SiLU(),
            nn.Conv2d(width, 4 * width, 3, padding=1),
            nn.PixelShuffle(2), nn.SiLU(),
            nn.Conv2d(width, 3, 3, padding=1), nn.Sigmoid(),
        )

    def encode(self, image):
        """输入 [N,3,H,W] 且像素在 [0,1]。返回 R/B/G 位流及特征空间尺寸。"""
        h, w = image.shape[-2:]
        padded = F.pad(image, (0, (-w) % 4, 0, (-h) % 4), mode="replicate")
        core = self.core_bits(self.core_encoder(padded))
        logits = torch.cat((core, self.encoder(padded)), dim=1)
        soft = torch.sigmoid(logits)
        hard = (logits >= 0).to(logits.dtype)
        quantized = soft + (hard - soft).detach() if self.training else hard
        return quantized.reshape(image.size(0), 3, -1), logits.shape[-2:]

    def classify(self, received, latent_size):
        """仅从恢复逻辑顺序后的 R 核心位预测十类 logits。"""
        red = received[:, 0].reshape(received.size(0), self.channels, *latent_size)
        return self.task_head(red)

    def decode(self, bits, latent_size, output_size, mode, snr_db):
        """解码时显式屏蔽停用色，附加三路可用性与归一化 SNR 四张条件图。"""
        h, w = output_size
        active = (1., float(mode >= 4), float(mode == 8))
        mask = bits.new_tensor(active).view(1, 3, 1)
        quality = (min(30., max(-5., snr_db)) + 5.) / 35.
        condition = bits.new_tensor((*active, quality)).view(1, 4, 1, 1)
        features = (bits * mask).reshape(bits.size(0), 3 * self.channels, *latent_size)
        features = torch.cat((features, condition.expand(bits.size(0), -1, *latent_size)), dim=1)
        return self.decoder(features)[:, :, :h, :w]

    def forward(self, image, link, snr_db, mode=8, noise=None):
        bits, shape = self.encode(image)
        received = link(bits, snr_db, mode, noise)
        reconstruction = self.decode(received.bits, shape, image.shape[-2:], mode, snr_db)
        return reconstruction, bits, received
````

````python
"""预算内的完整 JPEG 码流。像素按 RGB，连续比特按 R、B、G 轮流装入信息字。"""
import io
from dataclasses import dataclass

import numpy as np
import torch
from PIL import Image


@dataclass
class JPEGFrames:
    bits: torch.Tensor
    sizes: list        # 含 JPEG 头部的实际字节数，0 表示发端预算不足


def pack(images, symbols, qualities):
    """symbols 是每张图的信息字数 S，预算 floor(3S/8) 字节，空位补零。"""
    arrays = images.detach().clamp(0, 1).mul(255).round().byte().permute(0, 2, 3, 1).cpu().numpy()
    frames = np.zeros((len(arrays), symbols * 3), dtype=np.uint8)
    sizes = []
    for index, array in enumerate(arrays):
        image = Image.fromarray(array)
        size = 0
        for quality in qualities:
            stream = io.BytesIO()
            image.save(stream, format="JPEG", quality=quality, subsampling=2, optimize=True)
            payload = stream.getvalue()
            if len(payload) * 8 <= symbols * 3:
                size = len(payload)
                frames[index, :size * 8] = np.unpackbits(np.frombuffer(payload, np.uint8))
                break
        sizes.append(size)
    bits = torch.from_numpy(frames.reshape(-1, symbols, 3)).transpose(1, 2)
    return JPEGFrames(bits.to(device=images.device, dtype=images.dtype), sizes)


def unpack(bits, image_size):
    """接收端只使用固定帧预算和约定尺寸。无法解码时输出中灰，并计入图像指标。"""
    h, w = image_size
    flat = bits.transpose(1, 2).contiguous().byte().cpu().numpy().reshape(bits.shape[0], -1)
    payloads = np.packbits(flat[:, :flat.shape[1] // 8 * 8], axis=1)
    images = []
    failures = 0
    for payload in payloads:
        try:
            with Image.open(io.BytesIO(payload.tobytes()), formats=("JPEG",)) as image:
                if image.size != (w, h):
                    raise ValueError("帧尺寸错误")
                decoded = np.array(image.convert("RGB"), dtype=np.float32) / 255
        except (OSError, SyntaxError, ValueError, Image.DecompressionBombError):
            decoded = np.full((h, w, 3), 0.5, dtype=np.float32)
            failures += 1
        images.append(decoded)
    return torch.from_numpy(np.stack(images)).permute(0, 3, 1, 2).to(bits.device), failures
````

## baselines/jpeg_fec.py

````python
"""等时隙的 JPEG 保护传输。完整 JPEG 经无损 DEFLATE，再用 RM(1,4) 分组纠错。"""
import io
import zlib
from functools import lru_cache

import numpy as np
import torch
from PIL import Image

from baselines.jpeg_mcpam import JPEGFrames

# 32 个 5 位消息代表 GF(2)^4 上的 32 个仿射函数，真值表即 16 位码字。
CODE = np.array([[(int(m & x).bit_count() + (m >> 4)) % 2 for x in range(16)]
                 for m in range(32)], dtype=np.uint8)
POW5 = 1 << np.arange(4, -1, -1)


def jpeg_bytes(image, quality):
    stream = io.BytesIO()
    image.save(stream, format="JPEG", quality=quality, optimize=True, subsampling=2)
    return stream.getvalue()


@lru_cache(maxsize=1)
def dictionary():
    """收发两端固定的无损压缩字典，只使用纯黑 32×32 合成图，没有样本相关旁信息。"""
    blank = Image.new("RGB", (32, 32))
    return b"".join(jpeg_bytes(blank, q) for q in (95,90,85,80,75,70,60,50,40,30,20,10,5,1))[-32768:]


@lru_cache(maxsize=8)
def permutation(length):
    """打散同一 RM 码字的位置，降低同一 SIC 字内的相关错误。"""
    return np.random.default_rng(314159).permutation(length)


def rm_encode(data, capacity):
    """输入 [N,信息字节]，返回每张图恰好 capacity 个物理分支比特，尾部补零。"""
    blocks = capacity // 16
    payload = np.zeros((len(data), blocks*5), np.uint8)
    payload[:, :data.shape[1]*8] = np.unpackbits(data, axis=1)
    messages = payload.reshape(len(data), blocks, 5) @ POW5
    coded = np.zeros((len(data), capacity), np.uint8)
    coded[:, :blocks*16] = CODE[messages].reshape(len(data), -1)
    return coded[:, permutation(capacity)]


def rm_decode(physical):
    """硬比特上按最小汉明距离解码。16 点 Walsh 变换同时比较全部仿射函数。"""
    n, capacity = physical.shape
    blocks = capacity // 16
    ordered = np.empty_like(physical)
    ordered[:, permutation(capacity)] = physical
    signs = 1 - 2*ordered[:, :blocks*16].reshape(n, blocks, 16).astype(np.int16)
    for half in (1, 2, 4, 8):
        pairs = signs.reshape(n, blocks, -1, 2, half)
        left = pairs[..., 0, :].copy()
        right = pairs[..., 1, :].copy()
        pairs[..., 0, :] = left + right
        pairs[..., 1, :] = left - right
    slope = np.abs(signs).argmax(axis=-1)
    peak = np.take_along_axis(signs, slope[..., None], axis=-1)[..., 0]
    messages = slope + 16*(peak < 0)
    bits = ((messages[..., None] & POW5) != 0).reshape(n, -1)
    return np.packbits(bits[:, :blocks*5//8*8], axis=1)


def pack(images, symbols, qualities):
    """完整文件和最终发射帧均受 3S 位约束。384 字节帧有 120 信息字节（含 4 字节 CRC）。"""
    capacity = 3*symbols
    data = np.zeros((len(images), capacity//16*5//8), np.uint8)
    pixels = images.detach().clamp(0,1).mul(255).round().byte().permute(0,2,3,1).cpu().numpy()
    sizes = []
    for i, array in enumerate(pixels):
        image = Image.fromarray(array)
        size = 0
        for q in qualities:
            raw = jpeg_bytes(image, q)
            if len(raw)*8 > capacity:
                continue
            encoder = zlib.compressobj(9, zlib.DEFLATED, -15, zdict=dictionary())
            packed = encoder.compress(raw) + encoder.flush()
            if len(packed)+4 > data.shape[1]:
                continue
            size = len(raw)
            data[i, :len(packed)] = np.frombuffer(packed, np.uint8)
            crc = zlib.crc32(data[i, :-4]).to_bytes(4, "big")
            data[i, -4:] = np.frombuffer(crc, np.uint8)
            break
        sizes.append(size)
    coded = rm_encode(data, capacity)
    bits = torch.from_numpy(coded.reshape(len(images), symbols, 3)).transpose(1, 2)
    return JPEGFrames(bits.to(images), sizes)


def unpack(bits, image_size):
    """执行 RM 纠错、CRC 校验、限长解压和 JPEG 解码。丢帧按中灰图统计。"""
    h, w = image_size
    physical = bits.transpose(1,2).contiguous().byte().cpu().numpy().reshape(len(bits), -1)
    decoded = rm_decode(physical) if physical.shape[1] >= 16 else [b""]*len(bits)
    limit = physical.shape[1]//8
    images, failures = [], 0
    for row in decoded:
        try:
            data = bytes(row)
            if len(data)<5 or zlib.crc32(data[:-4]) != int.from_bytes(data[-4:], "big"):
                raise ValueError("CRC 校验失败")
            stream = zlib.decompressobj(-15, zdict=dictionary())
            raw = stream.decompress(data[:-4], limit+1)
            if not stream.eof or len(raw)>limit:
                raise ValueError("压缩载荷越界")
            with Image.open(io.BytesIO(raw), formats=("JPEG",)) as img:
                if img.size != (w,h):
                    raise ValueError("JPEG 尺寸错误")
                array = np.array(img.convert("RGB"), dtype=np.float32)/255
        except (ValueError, OSError, SyntaxError, zlib.error, Image.DecompressionBombError):
            array = np.full((h,w,3), .5, np.float32)
            failures += 1
        images.append(array)
    return torch.from_numpy(np.stack(images)).permute(0,3,1,2).to(bits.device), failures
````

````python
"""共享双任务编解码器训练。所有功率和交织对照共享源编码、接收网络和参数。"""
import json

def transmit(bits, link, mode, db, indices=None, noise=None):
    """交织只在全开消融中使用，解码器总是收到恢复逻辑顺序的特征。"""
    sent = bits if indices is None else bits.gather(1, indices)
    physical = link(sent, db, mode, noise).bits
    return physical if indices is None else restore(physical, indices)


def dual_loss(model, x, label, logical, shape, mode, db, weight):
    y = model.decode(logical, shape, x.shape[-2:], mode, db)
    mse = F.mse_loss(y, x)
    if label is None:
        return mse
    return mse + weight * F.cross_entropy(model.classify(logical, shape), label)


@torch.inference_mode()
def validate(model, links, loader, cfg, device):
    """按验证集两个任务的训练目标选权重，交织与有序固定模式均参与。"""
    model.eval()
    rng = torch.Generator().manual_seed(cfg.seed + 91)
    total = count = 0
    for item in loader:
        x, label = item if cfg.dataset == "cifar10" else (item, None)
        x = x.to(device)
        if label is not None:
            label = label.to(device)
        bits, shape = model.encode(x)
        index = color_indices(bits, rng)
        noise = torch.randn(len(x), links[0].slots_per_word, bits.size(2), generator=rng).to(device)
        for mode, db in cfg.val_points:
            for link, mapping in ((links[0], index), (links[0], None), (links[1], None)):
                operating_mode = 8 if link is links[0] else mode
                rx = transmit(bits, link.eval(), operating_mode, db, mapping, noise)
                loss = dual_loss(model, x, label, rx, shape, operating_mode, db, cfg.task_weight)
                total += loss.item() * len(x)
                count += len(x)
    return total / count


def fit(cfg):
    device = initialize(cfg)
    labeled = cfg.dataset == "cifar10"
    classifier, info = frozen(cfg, device, True) if labeled else (None, None)
    if info and (info["seed"], info["train_limit"], info["holdout"]) != (
            cfg.seed, cfg.train_limit, cfg.validation_holdout):
        raise ValueError("分类器与训练划分不一致，请使用独立 --run")
    initialize(cfg)
    train, val = datasets(cfg, True, labeled=labeled)
    batches = DataLoader(train, cfg.batch_size, shuffle=True)
    model = SemanticNet(cfg.channels, cfg.width).to(device)
    if classifier is not None:
        model.core_encoder.load_state_dict(classifier.features[:-1].state_dict())
    links = [MCPAMChannel(r, cfg.compression, cfg.balanced_pairs).to(device)
             for r in (cfg.uniform, cfg.proposed)]
    optimizer = torch.optim.AdamW(model.parameters(), lr=cfg.lr)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, cfg.epochs, eta_min=.03*cfg.lr)
    rng = np.random.default_rng(cfg.seed)
    interleave_rng = torch.Generator().manual_seed(cfg.seed + 13)
    best = float("inf")
    history = []
    for epoch in range(1, cfg.epochs + 1):
        model.train()
        loss_sum = 0.
        for step, item in enumerate(batches):
            x, label = item if labeled else (item, None)
            x = x.to(device)
            if label is not None:
                label = label.to(device)
            if rng.random() < .5:
                x = x.flip((-1,))
            if step % 3 == 0:
                order = rng.permutation(3)
            plan = order[step % 3]
            # 三组轮换：无序 4:2:1、有序 4:2:1、有序 6:2:1。降阶仅用于拟部署的自适应组。
            link = links[int(plan == 2)].train()
            mode = int(rng.choice((2, 4, 8), p=cfg.mode_probs)) if plan == 2 else 8
            db = float(rng.uniform(*cfg.train_snr))
            if rng.random() < .1:
                mode, db = 8, float("inf")
            bits, shape = model.encode(x)
            mapping = color_indices(bits, interleave_rng) if plan == 0 else None
            logical = transmit(bits, link, mode, db, mapping)
            loss = dual_loss(model, x, label, logical, shape, mode, db, cfg.task_weight)
            optimizer.zero_grad(set_to_none=True)
            loss.backward()
            optimizer.step()
            loss_sum += loss.item() * len(x)
        score = validate(model, links, DataLoader(val, cfg.batch_size), cfg, device)
        history.append(dict(epoch=epoch, loss=loss_sum / len(train), val_loss=score,
                            lr=optimizer.param_groups[0]["lr"]))
        scheduler.step()
        print(f"epoch {epoch}/{cfg.epochs} validation dual loss={score:.4f}", flush=True)
        if score < best:
            best = score
            save_best(cfg.directory("checkpoints") / "codec.pt",
                      dict(model=model.state_dict(), config=asdict(cfg), epoch=epoch,
                           val_loss=score, train_images=len(train), val_images=len(val), classifier=info))
        (cfg.directory("checkpoints") / "training.json").write_text(json.dumps(history, indent=2))


if __name__ == "__main__":
    fit(parameters(__doc__, True))
````

````python
def counts_as_dict(errors, totals):
    fields = {}
    for i, color in enumerate(COLORS):
        fields[f"errors_{color}"] = int(errors[i])
        fields[f"bits_{color}"] = int(totals[i])
        fields[f"ber_{color}"] = errors[i] / totals[i] if totals[i] else None
    return fields


@dataclass
class Accumulator:
    images: int = 0
    slots: int = 0
    psnr: float = 0.
    ssim: float = 0.
    optical: float = 0.
    errors: np.ndarray = field(default_factory=lambda: np.zeros(3, dtype=np.int64))
    totals: np.ndarray = field(default_factory=lambda: np.zeros(3, dtype=np.int64))
    jpeg_failures: int = 0
    jpeg_overflows: int = 0
    correct: int = 0
    task_hits: int = 0
    task_samples: int = 0
    core_errors: int = 0
    core_total: int = 0
    labeled: int = 0
    jpeg_bytes: int = 0

    def add(self, original, decoded, bits, received, slots_per_word, classifier, labels,
            logits=None, logical_core=None, decoded_core=None):
        """物理 BER 按发射颜色统计，逻辑 R 的 BER 在解交织后单独统计。"""
        if logical_core is not None:
            self.core_errors += (logical_core != decoded_core).sum().item()
            self.core_total += logical_core.numel()
        if labels is not None:
            self.correct += (classifier(decoded).argmax(1).cpu() == labels).sum().item()
            self.labeled += len(labels)
            if logits is not None:
                self.task_hits += (logits.argmax(1).cpu() == labels).sum().item()
                self.task_samples += len(labels)
        psnr, ssim = image_scores(original, decoded)
        errors, totals = bit_counts(bits, received)
        self.errors += errors
        self.totals += totals
        self.psnr += psnr.sum().item()
        self.ssim += ssim.sum().item()
        self.images += len(original)
        slots_per_image = slots_per_word * bits.size(2)
        self.slots += len(original) * slots_per_image
        self.optical += received.optical_mean.sum().item() * slots_per_image

    def row(self, scheme, snr, mode):
        jpeg = scheme in (JPEG, RAW_JPEG)
        task_acc = self.task_hits / self.task_samples if self.task_samples else None
        recon_acc = self.correct / self.labeled if self.labeled else None
        return dict(scheme=scheme, snr_db=snr, mode=mode, images=self.images,
                    physical_slots=self.slots, mean_optical_power=self.optical / self.slots,
                    acc=recon_acc if jpeg else task_acc, task_acc=task_acc, recon_acc=recon_acc,
                    task_correct=self.task_hits if self.task_samples else None,
                    recon_correct=self.correct if self.labeled else None,
                    core_errors=self.core_errors if self.core_total else None, core_bits=self.core_total,
                    core_ber=self.core_errors/self.core_total if self.core_total else None,
                    psnr=self.psnr / self.images, ssim=self.ssim / self.images,
                    jpeg_decode_failures=self.jpeg_failures if jpeg else None,
                    jpeg_success_rate=1-self.jpeg_failures/self.images if jpeg else None,
                    jpeg_budget_failures=self.jpeg_overflows if jpeg else None,
                    jpeg_mean_bytes=self.jpeg_bytes / self.images if jpeg else None,
                    **counts_as_dict(self.errors, self.totals))


def probe_ber(cfg, links, device):
    """使用同一组三路 IID 位流和噪声，独立比较权重本身的误码影响。"""
    gen = torch.Generator().manual_seed(cfg.seed + 23)
    shape = (1, 3, cfg.probe_symbols)
    bits = torch.randint(2, shape, generator=gen).float().to(device)
    noise = torch.randn(1, links["uniform"].slots_per_word, cfg.probe_symbols, generator=gen).to(device)
    rows = []
    for snr in cfg.snrs:
        for name, scheme in (("uniform", ORDERED), ("proposed", FIXED)):
            received = links[name](bits, snr, 8, noise)
            rows.append(dict(snr_db=snr, scheme=scheme, **counts_as_dict(*bit_counts(bits, received))))
    return rows



def write_rows(path, rows):
    with path.open("w", newline="", encoding="utf-8-sig") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)

def evaluate(cfg):

    path = cfg.directory("checkpoints") / "codec.pt"
    state = torch.load(path, map_location="cpu", weights_only=True)

    model = SemanticNet(training.channels, training.width).to(device).eval()
    model.load_state_dict(state.pop("model"))
    classifier = None
    if training.dataset == "cifar10" or cfg.dataset == "cifar10":
        classifier, info = frozen(cfg, device)

            raise ValueError("冻结分类器与训练记录不一致")
    calibration_dir = cfg.data_dir if training.dataset == cfg.dataset else training.data_dir
    reports, selected = calibrate(model, replace(training, snrs=cfg.snrs, data_dir=calibration_dir), device)
    if cfg.eval_ratio:
        wanted = tuple(float(x) for x in cfg.eval_ratio.split(":"))
        matches = [r for r in reports if tuple(r["ratio"]) == wanted]
        if not matches:
            raise ValueError(f"指定比例须属于训练候选 {training.candidates}")
        selected = matches[0]
    policy = selected["policy"]
    cfg = replace(cfg, proposed=selected["ratio"], uniform=training.uniform,
                  balanced_pairs=training.balanced_pairs, compression=training.compression)
    links = {name: MCPAMChannel(r, cfg.compression, cfg.balanced_pairs).eval().to(device)
             for name, r in (("uniform", cfg.uniform), ("proposed", cfg.proposed))}
    labeled = cfg.dataset == "cifar10"
    test = datasets(cfg, False, labeled=labeled)
    gen = torch.Generator().manual_seed(cfg.seed + 17)
    mapping_rng = torch.Generator().manual_seed(cfg.seed + 117)
    modes = (2, 4, 8)
    keys = (JPEG, RAW_JPEG, UNORDERED, ORDERED, *modes)
    sums = {(s, db): Accumulator() for s in keys for db in cfg.snrs}
    clean = {key: Accumulator() for key in (JPEG, RAW_JPEG, UNORDERED, ORDERED, 8)}
    originals_correct = 0
    comparisons = ((ORDERED, UNORDERED), (FIXED, ORDERED), (ADAPTIVE, UNORDERED))
    paired = {(db, a, b): [0, 0] for db in cfg.snrs for a, b in comparisons}
    examples = {}
    k = links["uniform"].slots_per_word
    for batch, item in enumerate(DataLoader(test, cfg.batch_size)):
        images, labels = item if labeled else (item, None)
        images = images.to(device)
        if labeled:
            originals_correct += (classifier(images).argmax(1).cpu() == labels).sum().item()
        bits, latent = model.encode(images)
        indices = color_indices(bits, mapping_rng)
        interleaved = bits.gather(1, indices)
        noise = torch.randn(len(images), k, bits.size(2), generator=gen).to(device)
        jpeg_codecs = {JPEG: jpeg_fec, RAW_JPEG: jpeg_mcpam}
        frames = {name: codec.pack(images, bits.size(2), cfg.jpeg_qualities)
                  for name, codec in jpeg_codecs.items()}
        for db in (*cfg.snrs, float("inf")):
            noiseless = db == float("inf")
            preview = {}
            hits = {}
            for key in (clean if noiseless else keys):
                acc = clean[key] if noiseless else sums[key, db]
                mode = key if isinstance(key, int) else 8
                sent = frames[key].bits if key in frames else interleaved if key == UNORDERED else bits
                link = links["proposed" if isinstance(key, int) else "uniform"]
                received = link(sent, db, mode, noise)
                if key in frames:
                    decoded, failures = jpeg_codecs[key].unpack(received.bits, images.shape[-2:])
                    acc.jpeg_failures += failures
                    acc.jpeg_overflows += frames[key].sizes.count(0)
                    acc.jpeg_bytes += sum(frames[key].sizes)
                    acc.add(images, decoded, sent, received, k, classifier, labels)
                else:
                    logical = restore(received.bits, indices) if key == UNORDERED else received.bits
                    decoded = model.decode(logical, latent, images.shape[-2:], mode, db)
                    logits = model.classify(logical, latent) if labeled else None
                    if labeled and not noiseless:
                        hits[key] = logits.argmax(1).cpu() == labels
                    acc.add(images, decoded, sent, received, k, classifier, labels,
                            logits, bits[:, 0], logical[:, 0])
                if batch == 0 and db in (5, 15):
                    preview[key] = decoded[0].cpu()
            if preview:
                examples[db] = [images[0].cpu(), preview[JPEG], preview[UNORDERED], preview[ORDERED],
                                preview[8], preview[active_mode(policy, db)]]
            if hits:
                hits[FIXED] = hits[8]
                hits[ADAPTIVE] = hits[active_mode(policy, db)]
                for a, b in comparisons:
                    # 同一测试图在两个任务头中的对错差异，保留配对胜负原始计数。
                    paired[db, a, b][0] += (hits[a] & ~hits[b]).sum().item()
                    paired[db, a, b][1] += (~hits[a] & hits[b]).sum().item()
    # 整个 SNR 点采用共同策略；复用已计算的同一模式结果，保持逐图逐噪声严格配对。
    rows, mode_rows = [], []
    for db in cfg.snrs:
        mode = active_mode(policy, db)
        for scheme, key in zip(SCHEMES, (JPEG, UNORDERED, ORDERED, 8, mode)):
            rows.append(sums[key, db].row(scheme, db, key if isinstance(key, int) else 8))
        mode_rows.extend(sums[m, db].row("fixed", db, m) for m in modes)
    if len({r["physical_slots"] for r in rows}) != 1 or (cfg.balanced_pairs and any(
            abs(r["mean_optical_power"] - .5) > 1e-6 for r in rows)):
        raise RuntimeError("物理时隙或平均功率不一致")
    probe = probe_ber(cfg, links, device)
    directory = Path(cfg.output_dir) if cfg.output_dir else cfg.directory("results")
    if cfg.eval_ratio and not cfg.output_dir:
        directory = directory / ("ratio_" + "_".join(f"{x:g}" for x in selected["ratio"]))
    directory.mkdir(parents=True, exist_ok=True)
    write_rows(directory / "metrics.csv", rows)
    write_rows(directory / "jpeg_uncoded.csv", [sums[RAW_JPEG, db].row(RAW_JPEG, db, 8) for db in cfg.snrs])
    write_rows(directory / "mode_metrics.csv", mode_rows)
    record = dict(config=asdict(cfg), checkpoint=state, device=str(device), torch=str(torch.__version__),
                  python=platform.python_version(), matplotlib=matplotlib.__version__, numpy=np.__version__,
                  pillow=PIL.__version__, test_images=len(test), validation_candidates=reports,
                  policy=policy, clean_original_top1=originals_correct/len(test) if labeled else None,
                  noiseless={name: clean[key].row(name, "noiseless", 8) for key, name in
                             ((JPEG, JPEG), (RAW_JPEG, RAW_JPEG), (UNORDERED, UNORDERED), (ORDERED, ORDERED), (8, FIXED))},
                  jpeg_transport=dict(source="complete JPEG", packing="raw DEFLATE, public black-image dictionary",
                                      correction="interleaved RM(1,4) [16,5,8]", checksum="CRC32",
                                      reference="jpeg_uncoded.csv"),
                  interleaver_seed=cfg.seed+117, iid_ber=probe,
                  paired_task_comparisons=[dict(snr_db=db, first=a, second=b,
                      first_only_correct=v[0], second_only_correct=v[1])
                      for (db, a, b), v in paired.items()] if labeled else [])
    (directory / "run.json").write_text(json.dumps(record, indent=2, ensure_ascii=False), encoding="utf-8")
    save_quality(rows, directory, cfg)
    save_ber(rows, directory)
    save_modes(rows, mode_rows, directory, selected["psnr_policy"])
    if examples:
        save_examples(examples, directory)
    print(f"ratio={cfg.proposed}, policy={policy}, test={len(test)} -> {directory}", flush=True)


if __name__ == "__main__":
    evaluate(parameters(__doc__, training=False))
````

````python
"""逐图 RGB PSNR、局部 Gaussian SSIM，以及活动物理分支误码计数。"""
import torch
import torch.nn.functional as F


def image_scores(original, reconstructed):
    """返回各图 PSNR(dB) 与 SSIM。动态范围 1，完全相同的图像 PSNR 为正无穷。"""
    mse = (original - reconstructed).square().mean(dim=(1, 2, 3))
    psnr = -10 * torch.log10(mse)
    window = min(11, *original.shape[-2:])
    window -= 1 - window % 2
    coords = torch.arange(window, device=original.device, dtype=original.dtype) - window // 2
    gaussian = torch.softmax(-coords.square() / (2 * 1.5 ** 2), dim=0)
    kernel = torch.outer(gaussian, gaussian).expand(3, 1, window, window).contiguous()
    fields = torch.cat((original, reconstructed, original.square(), reconstructed.square(),
                        original * reconstructed))
    mx, my, ex2, ey2, exy = F.conv2d(fields, kernel, groups=3).chunk(5)
    luminance = (2 * mx * my + 0.01 ** 2) / (mx.square() + my.square() + 0.01 ** 2)
    structure = (2 * (exy - mx * my) + 0.03 ** 2) / (
        (ex2 - mx.square()).clamp_min(0) + (ey2 - my.square()).clamp_min(0) + 0.03 ** 2)
    return psnr, (luminance * structure).mean(dim=(1, 2, 3))


def bit_counts(sent, received):
    """每色原始错误数和比较位数；关闭颜色两项均记 0，BER 留空。"""
    active = (received.weights > 0).long()
    errors = (sent != received.bits).sum(dim=(0, 2)) * active
    totals = active * (sent.shape[0] * sent.shape[2])
    return errors.tolist(), totals.tolist()
````

````python
def save_best(path, state):

    path.parent.mkdir(parents=True, exist_ok=True)

    finally:
````

````python
        self.assertEqual(frames.bits.shape, (4, 3, 1024))
        self.assertTrue(all(0 < size <= 384 for size in frames.sizes))
        self.assertEqual(jpeg_fec.unpack(frames.bits, (32,32))[1], 0)
        self.assertEqual(jpeg_fec.unpack(torch.zeros_like(frames.bits), (32,32))[1], 4)

    def test_ssim(self):
        x = torch.rand(2, 3, 32, 32)
        psnr, ssim = image_scores(x, x.clone())
        self.assertTrue(torch.isposinf(psnr).all())
        torch.testing.assert_close(ssim, torch.ones(2))
        zeros, ones = torch.zeros_like(x), torch.ones_like(x)
        psnr, ssim = image_scores(zeros, ones)
        torch.testing.assert_close(psnr, torch.zeros(2))
        torch.testing.assert_close(ssim, torch.full((2,), .0001 / 1.0001))


if __name__ == "__main__":
    unittest.main()
````

````markdown
## 跨层消融

| 曲线 | 源编码与映射 | 物理模式 | 端到端分类输出 |
|---|---|---|---|
| JPEG + FEC 4:2:1 | 完整 JPEG、无损压缩、CRC、RM 纠错与物理交织 | 固定 8 阶 | 冻结分类器读取接收图像 |
| Unordered semantic 4:2:1 | 共享 CNN，各逻辑层在三种颜色上平衡打散 | 固定 8 阶 | 解交织后的逻辑核心位 → 任务头 |
| Ordered semantic 4:2:1 | 同一 CNN，逻辑核心位全部映射到红光 | 固定 8 阶 | 解调后的红路位 → 同一任务头 |
| Proposed 6:2:1 fixed | 相同有序映射，改变光强比例 | 固定 8 阶 | 解调后的红路位 → 同一任务头 |
| Proposed 6:2:1 adaptive | 相同有序映射和比例，按验证策略选模式 | 2、4 或 8 阶 | 解调后的红路位 → 同一任务头 |

前三个语义固定模式的对照分别隔离了映射顺序与光强比例。它们使用同一张图的同一组源比特、相同网络参数和相同的物理噪声实现。每张图独立生成颜色轮转次序，任一逻辑层的 1,024 位在物理 R、B、G 各占 341 或 342 位。接收端复用共享伪随机种子和帧顺序解交织，假设帧同步理想。程序为交织器和 AWGN 使用独立随机流。真实硬件需要实现双方的一致帧计数。

## 编解码器与训练

核心编码分支用干净分类器的前三个卷积块初始化，保留前两次池化得到 8×8 特征，再生成 16 个二值通道。另一卷积分支生成蓝绿各 16 个通道。共享编解码器端到端训练，红路任务头只读取接收端的**逻辑核心位**。重建头联合使用三层特征、三路活动标志和 SNR 条件，关闭的层置零。训练目标为
`MSE(重建图, 原图) + 0.5 × CrossEntropy(核心任务 logits, CIFAR-10 标签)`。两个头均无原图旁路。分类标签不进入发射比特流。

训练使用固定的 15,000 张 CIFAR-10 图像，训练留出集中固定选 1,024 张验证。每三个批次轮换三种映射/权重组合，两种均匀组固定 8 阶，非均匀组抽样嵌套模式。训练使用水平翻转、−5 到 25 dB 的连续噪声、10% 无噪全开样本和 30 轮余弦退火。按验证集联合损失最小保存共享 codec。本次复用 v2 同划分的干净分类器作为初始化来源与冻结评估器。语义任务头在 v3 训练。最终测试使用 10,000 张独立测试图，测试标签仅用于计算指标。

v3 自适应模式由验证集的双任务联合损失决定，目标与训练一致。模式图还显示 v2 的验证 PSNR 策略和测试单模式上界。任务准确率与重建质量随同一实际策略测量，两者各自的最优模式可能不同。阈值来自 5 dB 间隔的采样，插值精度需更密的验证测点支持。

## 符号预算与 JPEG

32×32 RGB 原图有 24,576 位。每个逻辑层为 16×8×8 = 1,024 位，固定 8 阶共 3,072 位，即 384 字节，是原始位数的 1/8。每个信息字发送原码与反码两个时隙，所有方案各占每图 2,048 个物理时隙。每个活动模式权重和为 1，逐图时间平均光强为 0.5。LED 采用中点对称 tanh 标定，两个时隙独立加噪。横轴为共同参考电 SNR，单时隙噪声方差为 `0.25 × 10^(-SNR_dB/10)`。

主对照在候选质量中选择同时满足两个预算的最高档。完整独立 JPEG 文件（含所有头部）须不超过 384 字节。随后对完整文件做无损 DEFLATE，所用公共字典由固定纯黑合成图生成，收发两端事先约定并使用相同字典。每帧的压缩结果和填充合计 116 字节，附加 4 字节 CRC32，共 120 字节，即 960 位。每 5 位分配一个 16 位的一阶 Reed–Muller 码字，192 个码字正好占 3,072 位，然后按固定置换交织发送。没有重传。每个码字最多 3 处硬判决错误能保证纠正。Walsh 变换实现硬判决下的最近码字译码。

每个像素、质量选择及图像相关表均来自收到的码流。预置字典由公开的固定算法生成，独立于训练/测试图像。接收端经纠错、CRC、限长无损解压后恢复完整 JPEG，按约定尺寸解码。CRC 或解码失败时输出中灰图，保留在图像指标和准确率分母。裸 JPEG 完整文件也受 384 字节约束，允许在自身预算内独立选质量，结果另存于 `jpeg_uncoded.csv`。两种实现具有相同的物理时隙和噪声，图像源质量及接收纠错能力可不同。
````
